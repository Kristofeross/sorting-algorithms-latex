\addcontentsline{toc}{chapter}{Spis skrótów}

\begin{description}[leftmargin=3cm, style=multiline]
    \item[ILP] \textit{Instruction-Level Parallelism} -- równoległość na poziomie instrukcji
    \item[TLP] \textit{Thread-Level Parallelism} -- równoległość na poziomie wątków
    \item[SMT] \textit{Simultaneous Multithreading} -- jednoczesna wielowątkowość
    \item[GIL] \textit{Global Interpreter Lock} -- globalna blokada interpretera
    \item[RSS] \textit{Resident Set Size} -- rozmiar pamięci rezydentnej
\end{description}
